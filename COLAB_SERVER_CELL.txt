# Helper function for running commands
def run(cmd, desc=None):
    print(f"\n{'='*60}")
    print(desc or cmd)
    print(f"{'='*60}")
    result = subprocess.run(cmd, shell=True, text=True, capture_output=True)
    if result.stdout.strip():
        print(result.stdout)
    if result.returncode != 0:
        print(f"❌ Error: {result.stderr}")
        raise RuntimeError(f"Command failed: {cmd}")
    return result

def download_model(gdrive_id, target_dir, desc):
    print(f"\n📥 Downloading {desc}...")
    cmd = f"gdown --folder {gdrive_id} -O {target_dir}"
    run(cmd, f"Downloading {desc}")
    
    for root, dirs, files in os.walk(target_dir):
        for file in files:
            if file in ['model.pth', 'model.json']:
                src = os.path.join(root, file)
                dst = os.path.join(target_dir, file)
                if src != dst:
                    shutil.move(src, dst)
    
    for root, dirs, files in os.walk(target_dir, topdown=False):
        for dir_name in dirs:
            dir_path = os.path.join(root, dir_name)
            if not os.listdir(dir_path):
                os.rmdir(dir_path)
    
    print(f"✅ {desc} downloaded successfully")

# Process video endpoint - accepts file upload
@app.route('/process_video', methods=['POST'])
def process_video():
    try:
        print("\n📹 Received video processing request")
        
        # Check if file was uploaded
        if 'video' not in request.files:
            return jsonify({'error': 'No video file provided'}), 400
        
        video_file = request.files['video']
        
        if video_file.filename == '':
            return jsonify({'error': 'Empty file'}), 400
        
        # Save uploaded file temporarily
        temp_path = tempfile.mktemp(suffix='.mp4')
        video_file.save(temp_path)
        print(f"📁 Saved video to: {temp_path}")
        
        # Process with AI model
        print("🔄 Processing with AI model...")
        result = pipeline(temp_path)
        
        # Clean up temp file
        if os.path.exists(temp_path):
            os.remove(temp_path)
        
        print(f"✅ Transcription result: {result}")
        
        return jsonify({
            'success': True,
            'transcription': result,
            'device': str(device)
        })
        
    except Exception as e:
        print(f"❌ Processing error: {e}")
        return jsonify({'error': str(e), 'success': False}), 500

# Process video from base64 (for real-time webcam)
@app.route('/process_base64', methods=['POST'])
def process_base64():
    try:
        print("\n📹 Received base64 video processing request")
        
        data = request.get_json()
        
        if 'video_base64' not in data:
            return jsonify({'error': 'No video_base64 provided'}), 400
        
        # Decode base64 to video file
        video_base64 = data['video_base64']
        video_bytes = base64.b64decode(video_base64)
        
        # Save to temp file
        temp_path = tempfile.mktemp(suffix='.mp4')
        with open(temp_path, 'wb') as f:
            f.write(video_bytes)
        
        print(f"📁 Decoded and saved video to: {temp_path}")
        
        # Process with AI model
        print("🔄 Processing with AI model...")
        result = pipeline(temp_path)
        
        # Clean up temp file
        if os.path.exists(temp_path):
            os.remove(temp_path)
        
        print(f"✅ Transcription result: {result}")
        
        return jsonify({
            'success': True,
            'transcription': result,
            'device': str(device)
        })
        
    except Exception as e:
        print(f"❌ Processing error: {e}")
        return jsonify({'error': str(e), 'success': False}), 500
